cmake_minimum_required(VERSION 3.30.2)
project(bridge_host LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(bridge_core STATIC "${ROOT}/firmware/components/bridge_core/startup.cpp")
target_include_directories(bridge_core PUBLIC "${ROOT}/firmware/components/bridge_core/include")
set_target_properties(bridge_core PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(bridge_core PRIVATE -fno-rtti -fno-exceptions)
add_library(platform_policy STATIC "${ROOT}/firmware/components/platform_esp32/startup_policy.cpp")
target_include_directories(platform_policy PUBLIC "${ROOT}/firmware/components/platform_esp32/include")
target_link_libraries(platform_policy PUBLIC bridge_core)
set_target_properties(platform_policy PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
enable_testing()
foreach(name startup_tests negative_probe core_header_probe)
  add_executable(${name} "${name}.cpp")
  target_link_libraries(${name} PRIVATE platform_policy)
  set_target_properties(${name} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
endforeach()
set_target_properties(core_header_probe PROPERTIES CXX_STANDARD 11)
add_test(NAME startup_tests COMMAND startup_tests)
add_test(NAME core_header_probe COMMAND core_header_probe)
